package contexty
import("context";"testing";"sync")
type notifiedContext struct { context.Context; once sync.Once; checked chan struct{} }
func(c *notifiedContext) Err() error { err:=c.Context.Err(); c.once.Do(func(){close(c.checked)}); return err }
func TestReviewClearCanceledWhileWaiting(t *testing.T){
 // Arrange: simulate an in-flight CommitState holding the store mutex.
 s:=NewMemoryConversationStateStore(); c,cancel:=context.WithCancel(context.Background()); ctx:=&notifiedContext{Context:c,checked:make(chan struct{})}
 s.mu.Lock(); done:=make(chan error,1)
 go func(){done<-s.ClearState(ctx,"review",0)}()
 <-ctx.checked; cancel()
 // Act: release store only after request cancellation is observable.
 s.mu.Unlock(); err:=<-done
 // Assert: proof of current defect; canceled clear still publishes revision.
 got,_:=s.LoadState(context.Background(),"review")
 if err!=nil || got.Version()!=1 {t.Fatalf("err=%v version=%d",err,got.Version())}
 t.Logf("ClearState returned nil and published version=%d after ctx cancellation",got.Version())
}
